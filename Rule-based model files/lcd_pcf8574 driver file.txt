from machine import I2C
import time

MASK_RS = 0x01
MASK_EN = 0x04
MASK_BACKLIGHT = 0x08

class I2cLcd:
    def __init__(self, i2c, addr=0x27, cols=16, rows=2):
        self.i2c = i2c
        self.addr = addr
        self.cols = cols
        self.rows = rows
        self.backlight = MASK_BACKLIGHT
        self.init_lcd()

    def _write(self, data):
        self.i2c.writeto(self.addr, bytes([data]))

    def _pulse(self, data):
        self._write(data | MASK_EN)
        time.sleep_us(50)
        self._write(data & ~MASK_EN)
        time.sleep_us(50)

    def _write4bits(self, value, mode=0):
        data = (value & 0xF0) | self.backlight | mode
        self._write(data)
        self._pulse(data)

    def send(self, value, mode=0):
        # High nibble
        self._write4bits(value & 0xF0, mode)
        # Low nibble
        self._write4bits((value << 4) & 0xF0, mode)

    def cmd(self, command):
        self.send(command, 0)

    def char(self, char):
        self.send(ord(char), MASK_RS)

    def string(self, text):
        for ch in text:
            self.char(ch)

    def clear(self):
        self.cmd(0x01)
        time.sleep_ms(2)

    def move(self, row, col):
        row_offsets = [0x00, 0x40]
        self.cmd(0x80 | (row_offsets[row] + col))

    def init_lcd(self):
        time.sleep_ms(50)
        self._write4bits(0x30)
        self._write4bits(0x30)
        self._write4bits(0x30)
        self._write4bits(0x20)

        self.cmd(0x28)  # 4-bit, 2 line
        self.cmd(0x0C)  # Display ON
        self.cmd(0x01)  # Clear
        time.sleep_ms(2)
        self.cmd(0x06)  # Cursor move right
